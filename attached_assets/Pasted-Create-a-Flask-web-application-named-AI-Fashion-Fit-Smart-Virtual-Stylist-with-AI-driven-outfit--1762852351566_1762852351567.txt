import google.generativeai as genai, json, os genai.configure(api_key=os.getenv("GEMINI_API_KEY"))

def analyze_user_photo(img_path): model = genai.GenerativeModel("gemini-1.5-pro-latest") with open(img_path, "rb") as f: prompt = """ Describe this person in JSON: {"skin_tone": "Fair/Wheatish/Dark", "body_shape": "Slim/Average/Curvy"}. Only output JSON. """ resp = model.generate_content([prompt, f.read()]) return json.loads(resp.text)

Filtering logic:

def match_outfits(user): with open("outfits.json") as f: outfits = json.load(f) matched = [] for o in outfits: min_age, max_age = map(int, o["age_group"].split("-")) if ( user["gender"].lower() == o["gender"].lower() and user["skin_tone"] in o["skin_tones"] and user["body_shape"] in o["body_shapes"] and min_age <= user["age"] <= max_age ): matched.append(o) return matched[:5]

🧍 5. /tryon_preview → Virtual Try-On (AI Image Generation)

Use Gemini to combine user photo + outfit image and generate realistic output.

import requests, uuid, io

def create_virtual_tryon(user_img, outfit_img_url): model = genai.GenerativeModel("gemini-1.5-pro-latest") user_data = open(user_img, "rb").read() outfit_data = requests.get(outfit_img_url).content prompt = "Generate a realistic image of person (image 1) wearing outfit (image 2)." result = model.generate_content([prompt, user_data, outfit_data]) img_data = result.candidates[0].content.parts[0].inline_data.data out_path = f"static/generated/{uuid.uuid4().hex}.jpg" with open(out_path, "wb") as f: f.write(io.BytesIO(img_data).getbuffer()) return out_path
